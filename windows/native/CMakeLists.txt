cmake_minimum_required(VERSION 3.21)
project(TingTingNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
if(MSVC AND CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
 # Upstream MSVC ARM requires this switch; it selects the ARM backend and
 # does not apply the x86 -march=native compiler option.
 set(GGML_NATIVE ON CACHE BOOL "" FORCE)
 # Keep the distributed ARM binary at the baseline NEON/FMA level instead
 # of baking Azure's optional i8mm/dotprod instruction set into every PC.
 foreach(feature dotprod i8mm sve sme)
  set(GGML_MACHINE_SUPPORTS_${feature} FALSE CACHE BOOL "" FORCE)
 endforeach()
else()
 set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# A baseline x64 build works on older Windows CPUs; ARM64 uses native NEON.
foreach(flag AVX AVX2 AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 FMA F16C BMI2)
 set(GGML_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(source-timed-whisper ../../launcher/native/SourceTimedWhisper.cpp app.manifest)
target_link_libraries(source-timed-whisper PRIVATE whisper)
add_library(sherpa-api SHARED IMPORTED)
set_target_properties(sherpa-api PROPERTIES IMPORTED_IMPLIB "${SHERPA_IMPLIB}" IMPORTED_LOCATION "${SHERPA_DLL}" INTERFACE_INCLUDE_DIRECTORIES "${SHERPA_INCLUDE}")
add_executable(speaker-embeddings ../../launcher/native/SpeakerEmbeddings.cpp app.manifest)
target_link_libraries(speaker-embeddings PRIVATE sherpa-api psapi)
if(MSVC)
 target_compile_options(source-timed-whisper PRIVATE /utf-8)
 target_compile_options(speaker-embeddings PRIVATE /utf-8)
endif()
